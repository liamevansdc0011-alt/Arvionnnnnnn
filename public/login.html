document.getElementById('loginForm').addEventListener('submit', async (e) => { e.preventDefault(); const username = e.target.username.value; const password = e.target.password.value; const res = await fetch('/login', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ username, password }) }); const data = await res.json(); if (data.success) { // Successful login ke baad launcher page par redirect window.location.href = data.redirectUrl || '/launcher'; } else { alert(data.message || 'Invalid credentials'); } });
